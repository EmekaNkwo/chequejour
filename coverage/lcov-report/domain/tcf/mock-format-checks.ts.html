function bandOrderProblems(label: string, bands: readonly string[]): string[] {
  const ranks = bands.map((b) => CEFR_BANDS.indexOf(b as (typeof CEFR_BANDS)[number]));
  const firstDrop = ranks.findIndex((r, i) => i > 0 && r < ranks[i - 1]);
